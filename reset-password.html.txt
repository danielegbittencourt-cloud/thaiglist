  <script>
    const SUPABASE_URL = "https://jjybnidhwfimcdwygafk.supabase.co";
    const SUPABASE_KEY = "sb_publishable_DyikY0WSJJ4apDc0_ZwfOw_x6LvfqM-";

    const db = window.supabase.createClient(
      SUPABASE_URL,
      SUPABASE_KEY
    );

    const form = document.getElementById("reset-form");
    const button = document.getElementById("save-button");
    const message = document.getElementById("message");

    async function checkAccess() {
      const { data, error } = await db.auth.getSession();

      if (error || !data.session) {
        message.textContent =
          "Link inválido ou expirado. Solicite uma nova recuperação de senha.";
        button.disabled = true;
        return;
      }

      message.textContent = "Acesso confirmado! Escolha sua nova senha.";
      button.textContent = "Salvar nova senha";
      button.disabled = false;
    }

    form.addEventListener("submit", async (event) => {
      event.preventDefault();

      const password = document.getElementById("password").value;
      const confirmation =
        document.getElementById("confirm-password").value;

      if (password.length < 8) {
        message.textContent = "A senha precisa ter pelo menos 8 caracteres.";
        return;
      }

      if (password !== confirmation) {
        message.textContent = "As senhas não são iguais.";
        return;
      }

      button.disabled = true;
      message.textContent = "Atualizando sua senha...";

      const { error } = await db.auth.updateUser({
        password: password
      });

      if (error) {
        message.textContent = "Erro: " + error.message;
        button.disabled = false;
        return;
      }

      message.textContent = "Senha atualizada com sucesso! 🌷";

      await db.auth.signOut();

      setTimeout(() => {
        window.location.href = "./";
      }, 2500);
    });

    checkAccess();
  </script>
